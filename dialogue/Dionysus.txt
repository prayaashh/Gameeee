name: Dionysus

# Where the conversation starts. Checked top to bottom, first match wins.
# (Later stages first, so a finished quest is checked before an active one.)
entry: flag lantern_done -> 4
entry: flag lantern_quest -> 3
entry: flag lantern_declined -> 5
entry: else -> 0

Stage 0: (default stage - first time the player talks to him)
	play: idle
	speak: "Oh, so you have come"
	goto: 6


Stage 1: (player said yes)
	play: talk
	speak: "Then go. The cave beyond the castle swallows everything."
	set: lantern_quest

Stage 2: (player said no)
	play: idle
	speak: "Then we have nothing more to say."
	set: lantern_declined

Stage 3: (quest accepted, not finished)
	play: idle
	speak: "Have you found it yet?"
	speak: "It is still down there, in the dark."

Stage 4: (quest finished)
	play: talk
	speak: "You found it. I knew the dark could not keep it from you."

Stage 5: (player said no, then came back)
	play: idle
	speak: "Changed your mind?"
	goto: 6

Stage 6:
	ask: "will you find my lantern?"
	choices: "yes" -> 1, "no" -> 2
